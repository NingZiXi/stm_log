cmake_minimum_required(VERSION 3.22)
project(stm_log_tests C CXX)
enable_testing()
if(MINGW)
    # 旧 MinGW 默认 CRT 的 vsnprintf 截断语义不是 C99，测试使用标准实现。
    add_compile_definitions(__USE_MINGW_ANSI_STDIO=1)
endif()

# 同一真实实现覆盖配置组合，不使用 MCU/HAL 替身。
foreach(variant default plain fileline disabled small)
    add_executable(log_${variant} test_log.c ../stm_log.c)
    target_include_directories(log_${variant} PRIVATE ..)
    target_compile_features(log_${variant} PRIVATE c_std_11)
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(log_${variant} PRIVATE -Wall -Wextra -Werror)
    endif()
    add_test(NAME ${variant} COMMAND log_${variant})
endforeach()
target_compile_definitions(log_plain PRIVATE STM_LOG_USE_COLORS=0 STM_LOG_AUTO_NEWLINE=0 STM_LOG_EARLY_BUFFER_SIZE=0)
target_compile_definitions(log_fileline PRIVATE STM_LOG_INCLUDE_FILE_LINE=1)
target_compile_definitions(log_disabled PRIVATE STM_LOG_ENABLED=0)
target_compile_definitions(log_small PRIVATE STM_LOG_BUFFER_SIZE=32)

add_subdirectory(.. log_core)
add_executable(log_cpp test_headers.cpp)
target_link_libraries(log_cpp PRIVATE stm_log)
target_compile_features(log_cpp PRIVATE cxx_std_17)
add_test(NAME cpp_headers COMMAND log_cpp)
add_executable(log_example ../example/main.c)
target_link_libraries(log_example PRIVATE stm_log)
add_test(NAME example COMMAND log_example)
